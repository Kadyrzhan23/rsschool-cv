function getTotalCashPrize(prizeFund, correctAnswers, playerActions) {
    let money = 0;
    let safeMoney = 0;
    let lifelines = 0;

    for (let i = 0; i < playerActions.length; i++) {
        let action = playerActions[i];

        for (let j = 0; j < action.length - 1; j++) {
            if (action[j] === "1" || action[j] === "2" || action[j] === "3") {
                lifelines++;
            }
        }

        let answer = action[action.length - 1];

        if (answer === "W") {
            return [money, lifelines];
        }

        if (answer === "X") {
            return [safeMoney, lifelines];
        }

        if (answer === correctAnswers[i]) {
            money += prizeFund[i];

            if ((i + 1) % 5 === 0) {
                safeMoney = money;
            }
        } else {
            return [0, lifelines];
        }
    }

    return [money, lifelines];
}
